set_source_files_properties(qml/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)

# What the app remembers (settings, recent files, window place), recovery
# sessions, Home's preview sources and the license texts it shows: Qt Core
# only, so tests/test_uistate needs no GUI.
add_library(openshape_uistate STATIC
    AppSettings.cpp AppSettings.h
    Licenses.cpp Licenses.h
    RecoverySession.cpp RecoverySession.h
    ThumbnailSource.cpp ThumbnailSource.h
)
target_link_libraries(openshape_uistate PUBLIC openshape_io Qt6::Core)
target_include_directories(openshape_uistate PUBLIC ${PROJECT_SOURCE_DIR}/src)
openshape_set_warnings(openshape_uistate)
# The license texts and notices of About -> Licenses (resources/licenses/,
# generated by scripts/licenses/licenses.py; docs/LICENSING.md), at
# :/openshape/licenses/. Re-run CMake's configure step after regenerating
# them when files were added or removed.
file(GLOB_RECURSE OPENSHAPE_LICENSE_FILES CONFIGURE_DEPENDS
    ${PROJECT_SOURCE_DIR}/resources/licenses/*.json
    ${PROJECT_SOURCE_DIR}/resources/licenses/*.txt)
qt_add_resources(openshape_uistate "openshape_licenses"
    PREFIX "/openshape"
    BASE ${PROJECT_SOURCE_DIR}/resources
    FILES ${OPENSHAPE_LICENSE_FILES}
)

qt_add_library(openshape_ui STATIC)
qt_add_qml_module(openshape_ui
    URI OpenShape
    VERSION 1.0
    OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/qml/OpenShape
    QML_FILES
        qml/Main.qml
        qml/ValueChip.qml
        qml/SketchOverlay.qml
        qml/HistoryPanel.qml
        qml/HelpOverlay.qml
        qml/Panel.qml
        qml/ActionButton.qml
        qml/Separator.qml
        qml/SectionLabel.qml
        qml/Theme.qml
        qml/AxisTriad.qml
        qml/AboutOverlay.qml
        qml/LicensesOverlay.qml
        qml/PreferencesOverlay.qml
        qml/RecoveryOverlay.qml
        qml/UnsavedOverlay.qml
        qml/HomeScreen.qml
        qml/ScrollRow.qml
        qml/SaveNameOverlay.qml
    SOURCES
        AppController.cpp AppController.h
        ThumbnailProvider.cpp ThumbnailProvider.h
        ViewportItem.cpp ViewportItem.h
)
target_link_libraries(openshape_ui PUBLIC openshape_render openshape_io openshape_uistate Qt6::Quick Qt6::QuickControls2)
target_include_directories(openshape_ui PUBLIC ${PROJECT_SOURCE_DIR}/src)
openshape_set_warnings(openshape_ui)
